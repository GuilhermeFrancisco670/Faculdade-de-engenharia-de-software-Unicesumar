-- vai ser criado a tabela endereço no banco de dados onibusescola
CREATE TABLE endereco ( 
  id_endereco  INT           AUTO_INCREMENT PRIMARY KEY, /* Criando a coluna de endereços que vai ser uma chave primaria que se auto incrementa*/
  cep          CHAR(8)       NOT NULL, 
  rua          VARCHAR(100)  NOT NULL,
  numero_casa  VARCHAR(10)   NOT NULL,
  bairro       VARCHAR(60)   NULL,
  cidade       VARCHAR(60)   NOT NULL,
  uf           CHAR(2)       NOT NULL,
  descricao    VARCHAR(150)  NULL,
  CONSTRAINT chk_endereco_cep CHECK (CHAR_LENGTH(cep) = 8)
) ENGINE = InnoDB;

CREATE TABLE escola (
  id_escola        INT          AUTO_INCREMENT PRIMARY KEY,
  nome             VARCHAR(100) NOT NULL,
  id_endereco      INT          NOT NULL,
  quantidade_aluno INT          NOT NULL DEFAULT 0,
  turno            ENUM('MANHA','TARDE','NOITE','INTEGRAL') NOT NULL,
  horario_entrada  TIME         NOT NULL,
  horario_saida    TIME         NOT NULL,
  publico          CHAR(1)      NOT NULL DEFAULT 'S',
  CONSTRAINT chk_escola_publico  CHECK (publico IN ('S','N')),
  CONSTRAINT chk_escola_qtd      CHECK (quantidade_aluno >= 0),
  CONSTRAINT chk_escola_horario  CHECK (horario_saida > horario_entrada)
) ENGINE = InnoDB;

CREATE TABLE responsavel (
  id_responsavel  INT          AUTO_INCREMENT PRIMARY KEY,
  nome            VARCHAR(100) NOT NULL,
  numero_celular  VARCHAR(15)  NOT NULL,
  id_endereco     INT          NOT NULL,
  horario_em_casa TIME         NULL,
  CONSTRAINT uq_responsavel_celular UNIQUE (numero_celular)
) ENGINE = InnoDB;

CREATE TABLE onibus (
  id_onibus  INT     AUTO_INCREMENT PRIMARY KEY,
  placa      CHAR(7) NOT NULL,
  capacidade INT     NOT NULL,
  CONSTRAINT uq_onibus_placa   UNIQUE (placa),
  CONSTRAINT chk_onibus_cap    CHECK (capacidade > 0)
) ENGINE = InnoDB;

/* mud_onibus do modelo original virou a FK id_onibus:
   trocar o motorista de onibus = atualizar essa coluna     */
CREATE TABLE motorista (
  id_motorista INT          AUTO_INCREMENT PRIMARY KEY,
  nome         VARCHAR(100) NOT NULL,
  cnh          CHAR(11)     NOT NULL,
  telefone     VARCHAR(15)  NULL,
  id_onibus    INT          NULL,
  CONSTRAINT uq_motorista_cnh UNIQUE (cnh)
) ENGINE = InnoDB;

CREATE TABLE rota (
  id_rota       INT          AUTO_INCREMENT PRIMARY KEY,
  descricao     VARCHAR(150) NOT NULL,
  turno         ENUM('MANHA','TARDE','NOITE') NOT NULL,
  horario_saida TIME         NOT NULL,
  id_motorista  INT          NOT NULL,
  id_escola     INT          NOT NULL
) ENGINE = InnoDB;

CREATE TABLE aluno (
  id_aluno        INT          AUTO_INCREMENT PRIMARY KEY,
  nome            VARCHAR(100) NOT NULL,
  data_nascimento DATE         NULL,
  turno           ENUM('MANHA','TARDE','NOITE') NOT NULL,
  id_escola       INT          NOT NULL,
  id_responsavel  INT          NOT NULL,
  id_endereco     INT          NOT NULL
) ENGINE = InnoDB;

/* aluno x rota e' N:N -> tabela associativa */
CREATE TABLE aluno_rota (
  id_aluno       INT NOT NULL,
  id_rota        INT NOT NULL,
  ordem_embarque INT NULL,
  PRIMARY KEY (id_aluno, id_rota)
) ENGINE = InnoDB;

CREATE TABLE log_matricula (

-- aluno + escola + responsavel
SELECT a.nome        AS aluno,
       e.nome        AS escola,
       r.nome        AS responsavel,
       r.numero_celular
FROM   aluno a
JOIN   escola e      ON e.id_escola      = a.id_escola
JOIN   responsavel r ON r.id_responsavel = a.id_responsavel
ORDER  BY e.nome, a.nome;

-- aluno + endereco (onde o onibus busca cada aluno)
SELECT a.nome AS aluno,
       CONCAT(en.rua,', ',en.numero_casa,' - ',en.bairro) AS ponto_de_embarque,
       en.cep
FROM   aluno a
JOIN   endereco en ON en.id_endereco = a.id_endereco
ORDER  BY en.rua, en.numero_casa;

-- rota + motorista + onibus + escola
SELECT r.descricao       AS rota,
       r.horario_saida,
       m.nome            AS motorista,
       o.placa,
       o.capacidade,
       e.nome            AS escola
FROM   rota r
JOIN   motorista m ON m.id_motorista = r.id_motorista
LEFT   JOIN onibus o ON o.id_onibus  = m.id_onibus
JOIN   escola e    ON e.id_escola    = r.id_escola;

-- ocupacao de cada rota (5 tabelas + agregacao)
SELECT r.descricao          AS rota,
       o.placa,
       o.capacidade,
       COUNT(ar.id_aluno)   AS alunos,
       o.capacidade - COUNT(ar.id_aluno) AS vagas
FROM   rota r
JOIN   motorista m  ON m.id_motorista = r.id_motorista
JOIN   onibus o     ON o.id_onibus    = m.id_onibus
LEFT   JOIN aluno_rota ar ON ar.id_rota = r.id_rota
GROUP  BY r.id_rota, r.descricao, o.placa, o.capacidade
ORDER  BY vagas;

-- alunos sem rota atribuida
SELECT a.nome AS aluno, e.nome AS escola
FROM   aluno a
JOIN   escola e ON e.id_escola = a.id_escola
LEFT   JOIN aluno_rota ar ON ar.id_aluno = a.id_aluno
WHERE  ar.id_aluno IS NULL;

-- irmaos: responsaveis com mais de um filho
SELECT r.nome AS responsavel,
       COUNT(a.id_aluno) AS filhos,
       GROUP_CONCAT(a.nome ORDER BY a.nome) AS nomes
FROM   responsavel r
JOIN   aluno a ON a.id_responsavel = r.id_responsavel
GROUP  BY r.id_responsavel, r.nome
HAVING COUNT(a.id_aluno) > 1;

-- 5.1 mantem escola.quantidade_aluno sempre correto
CREATE TRIGGER trg_aluno_ai
AFTER INSERT ON aluno
FOR EACH ROW
BEGIN
  UPDATE escola
     SET quantidade_aluno = quantidade_aluno + 1
   WHERE id_escola = NEW.id_escola;

  INSERT INTO log_matricula (id_aluno, id_escola, acao)
  VALUES (NEW.id_aluno, NEW.id_escola, 'MATRICULA');

CREATE TRIGGER trg_aluno_ad
AFTER DELETE ON aluno
FOR EACH ROW
BEGIN
  UPDATE escola
     SET quantidade_aluno = quantidade_aluno - 1
   WHERE id_escola = OLD.id_escola;

  INSERT INTO log_matricula (id_aluno, id_escola, acao)
  VALUES (OLD.id_aluno, OLD.id_escola, 'CANCELAMENTO');
END$$

-- 5.2 transferencia de escola: ajusta os dois contadores
CREATE TRIGGER trg_aluno_au
AFTER UPDATE ON aluno
FOR EACH ROW
BEGIN
  IF NEW.id_escola <> OLD.id_escola THEN
    UPDATE escola SET quantidade_aluno = quantidade_aluno - 1
      WHERE id_escola = OLD.id_escola;
    UPDATE escola SET quantidade_aluno = quantidade_aluno + 1
      WHERE id_escola = NEW.id_escola;

    INSERT INTO log_matricula (id_aluno, id_escola, acao)
    VALUES (NEW.id_aluno, NEW.id_escola, 'TRANSFERENCIA');
  END IF;
END$$

-- 5.3 impede embarcar aluno acima da capacidade do onibus
CREATE TRIGGER trg_aluno_rota_bi
BEFORE INSERT ON aluno_rota
FOR EACH ROW
BEGIN
  DECLARE v_capacidade INT DEFAULT 0;
  DECLARE v_ocupados   INT DEFAULT 0;

  SELECT IFNULL(o.capacidade, 0) INTO v_capacidade
  FROM   rota r
  JOIN   motorista m ON m.id_motorista = r.id_motorista
  LEFT   JOIN onibus o ON o.id_onibus  = m.id_onibus
  WHERE  r.id_rota = NEW.id_rota;

  SELECT COUNT(*) INTO v_ocupados
  FROM   aluno_rota
  WHERE  id_rota = NEW.id_rota;

  IF v_ocupados >= v_capacidade THEN
    SIGNAL SQLSTATE '45000'
      SET MESSAGE_TEXT = 'Onibus lotado: capacidade da rota atingida.';
  END IF;
END$$

-- 6.1 matricula o aluno e ja o coloca em uma rota
CREATE PROCEDURE sp_matricular_aluno (
  IN  p_nome           VARCHAR(100),
  IN  p_nascimento     DATE,
  IN  p_turno          VARCHAR(10),
  IN  p_id_escola      INT,
  IN  p_id_responsavel INT,
  IN  p_id_endereco    INT,
  IN  p_id_rota        INT,
  OUT p_id_aluno       INT
)
BEGIN
  DECLARE EXIT HANDLER FOR SQLEXCEPTION
  BEGIN
    ROLLBACK;
    RESIGNAL;
  END;

  IF NOT EXISTS (SELECT 1 FROM escola WHERE id_escola = p_id_escola) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Escola inexistente.';
  END IF;

  IF NOT EXISTS (SELECT 1 FROM responsavel
                  WHERE id_responsavel = p_id_responsavel) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Responsavel inexistente.';
  END IF;

  IF p_id_rota IS NOT NULL
     AND NOT EXISTS (SELECT 1 FROM rota WHERE id_rota = p_id_rota) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Rota inexistente.';
  END IF;

  START TRANSACTION;

    INSERT INTO aluno (nome, data_nascimento, turno,
                       id_escola, id_responsavel, id_endereco)
    VALUES (p_nome, p_nascimento, p_turno,
            p_id_escola, p_id_responsavel, p_id_endereco);

    SET p_id_aluno = LAST_INSERT_ID();

    IF p_id_rota IS NOT NULL THEN
      INSERT INTO aluno_rota (id_aluno, id_rota, ordem_embarque)
      VALUES (p_id_aluno, p_id_rota,
              (SELECT IFNULL(MAX(ordem_embarque),0) + 1
                 FROM aluno_rota WHERE id_rota = p_id_rota));
    END IF;

  COMMIT;
END$$

-- 6.2 lista de embarque de uma rota (para o motorista)
CREATE PROCEDURE sp_lista_embarque (IN p_id_rota INT)
BEGIN
  SELECT ar.ordem_embarque       AS ordem,
         a.nome                  AS aluno,
         CONCAT(en.rua,', ',en.numero_casa) AS endereco,
         en.bairro,
         r.nome                  AS responsavel,
         r.numero_celular,
         e.nome                  AS escola,
         ro.horario_saida
  FROM   aluno_rota ar
  JOIN   rota  ro      ON ro.id_rota       = ar.id_rota
  JOIN   aluno a       ON a.id_aluno       = ar.id_aluno
  JOIN   endereco en   ON en.id_endereco   = a.id_endereco
  JOIN   responsavel r ON r.id_responsavel = a.id_responsavel
  JOIN   escola e      ON e.id_escola      = a.id_escola
  WHERE  ar.id_rota = p_id_rota
  ORDER  BY ar.ordem_embarque;
END$$

-- 6.3 troca o motorista de onibus (o antigo mud_onibus)
-- O mesmo onibus pode ser compartilhado por varios motoristas.
CREATE PROCEDURE sp_trocar_onibus (
  IN p_id_motorista INT,
  IN p_id_onibus    INT
)
BEGIN
  IF NOT EXISTS (SELECT 1 FROM onibus WHERE id_onibus = p_id_onibus) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Onibus inexistente.';
  END IF;

  UPDATE motorista
     SET id_onibus = p_id_onibus
   WHERE id_motorista = p_id_motorista;
END$$

--  Este script NAO recria o banco. Ele PARTE do banco do Trabalho 1

--    - 1 tabela de apoio      (alerta_rota)
--    - 3 Views                (organizam e preparam dados para processamento)
--    - 1 Stored Procedure     (chamada manualmente pelo usuario/coordenador)
--    - 1 Stored Procedure     (chamada de dentro de uma Trigger)
--    - 1 Trigger              (dispara a 2a procedure automaticamente)
--
--  PROBLEMA QUE ESTE INCREMENTO RESOLVE:
--  No Trabalho 1 sabemos QUAL onibus atende cada rota e QUANTOS alunos
--  estao nele, mas ninguem e avisado quando uma rota fica cheia.
--  O Trabalho 2 cria uma "central de ocupacao": sempre que um aluno entra
--  numa rota (evento no banco), o sistema calcula automaticamente a
--  ocupacao daquela rota e grava um alerta. Alem disso, o coordenador
--  pode rodar uma verificacao manual de todas as rotas quando quiser.

USE onibusescola;


-- -------------------------------------------------------------------------
-- 0) TABELA DE APOIO: guarda o efeito persistente gerado pelas procedures
-- -------------------------------------------------------------------------
CREATE TABLE alerta_rota (
  id_alerta       INT AUTO_INCREMENT PRIMARY KEY,
  id_rota         INT NOT NULL,
  nivel_alerta    ENUM('NORMAL','QUASE_LOTADO','LOTADO') NOT NULL,
  ocupacao_atual  INT NOT NULL,
  capacidade      INT NOT NULL,
  percentual      DECIMAL(5,2) NOT NULL,
  origem          ENUM('MANUAL','TRIGGER') NOT NULL,
  data_hora       DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_alerta_rota
      FOREIGN KEY (id_rota) REFERENCES rota (id_rota)
      ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE = InnoDB;


-- -------------------------------------------------------------------------
-- 1) VIEW 1 - vw_ocupacao_rota
--    Para que serve: junta rota + motorista + onibus + contagem de alunos
--    numa unica consulta pronta, calculando vagas e percentual de ocupacao.
--    E a "base de calculo" usada pelas duas Stored Procedures novas.
-- -------------------------------------------------------------------------
CREATE VIEW vw_ocupacao_rota AS
SELECT
    r.id_rota,
    r.descricao                                   AS rota,
    r.turno,
    m.nome                                         AS motorista,
    o.id_onibus,
    o.placa,
    o.capacidade,
    COUNT(ar.id_aluno)                             AS alunos_embarcados,
    (o.capacidade - COUNT(ar.id_aluno))            AS vagas_disponiveis,
    ROUND(COUNT(ar.id_aluno) / o.capacidade * 100, 2) AS percentual_ocupacao
FROM   rota r
JOIN   motorista m ON m.id_motorista = r.id_motorista
JOIN   onibus    o ON o.id_onibus    = m.id_onibus
LEFT   JOIN aluno_rota ar ON ar.id_rota = r.id_rota
GROUP  BY r.id_rota, r.descricao, r.turno, m.nome, o.id_onibus, o.placa, o.capacidade;


-- -------------------------------------------------------------------------
-- 2) VIEW 2 - vw_ficha_embarque
--    Para que serve: reune, por aluno e por rota, o endereco de embarque,
--    o responsavel/telefone e a escola de destino. Prepara em um so lugar
--    os dados que o motorista/coordenador precisa consultar todo dia,
--    sem repetir o JOIN de 5 tabelas toda vez.
-- -------------------------------------------------------------------------
CREATE VIEW vw_ficha_embarque AS
SELECT
    ar.id_rota,
    ro.descricao                                            AS rota,
    ro.horario_saida,
    ar.ordem_embarque,
    a.id_aluno,
    a.nome                                                   AS aluno,
    CONCAT(en.rua, ', ', en.numero_casa,
           IFNULL(CONCAT(' - ', en.bairro), ''))              AS endereco_embarque,
    en.cep,
    resp.nome                                                 AS responsavel,
    resp.numero_celular,
    esc.nome                                                  AS escola
FROM   aluno_rota ar
JOIN   rota        ro   ON ro.id_rota        = ar.id_rota
JOIN   aluno        a   ON a.id_aluno        = ar.id_aluno
JOIN   endereco     en  ON en.id_endereco    = a.id_endereco
JOIN   responsavel  resp ON resp.id_responsavel = a.id_responsavel
JOIN   escola       esc ON esc.id_escola     = a.id_escola;


-- -------------------------------------------------------------------------
-- 3) VIEW 3 - vw_alunos_sem_rota
--    Para que serve: lista os alunos matriculados que ainda NAO foram
--    colocados em nenhuma rota de onibus. Ajuda o coordenador a organizar
--    a proxima consulta/tarefa (quem falta encaixar em uma rota).
-- -------------------------------------------------------------------------
CREATE VIEW vw_alunos_sem_rota AS
SELECT
    a.id_aluno,
    a.nome              AS aluno,
    a.turno,
    esc.nome            AS escola,
    resp.nome           AS responsavel,
    resp.numero_celular
FROM   aluno a
JOIN   escola      esc  ON esc.id_escola      = a.id_escola
JOIN   responsavel resp ON resp.id_responsavel = a.id_responsavel
LEFT   JOIN aluno_rota ar ON ar.id_aluno = a.id_aluno
WHERE  ar.id_aluno IS NULL;


DELIMITER $$

-- -------------------------------------------------------------------------
-- 4) STORED PROCEDURE 2 (a que fica "dentro" da Trigger)
--    sp_verificar_ocupacao_rota
--    Quando executa: toda vez que a Trigger dispara (nunca e chamada
--    diretamente pelo usuario).
--    O que faz: le a View vw_ocupacao_rota para a rota informada,
--    decide o nivel do alerta e grava (efeito persistente) uma linha em
--    alerta_rota com origem = 'TRIGGER'.
-- -------------------------------------------------------------------------
CREATE PROCEDURE sp_verificar_ocupacao_rota (IN p_id_rota INT)
BEGIN
  DECLARE v_ocupados   INT;
  DECLARE v_capacidade INT;
  DECLARE v_percentual DECIMAL(5,2);
  DECLARE v_nivel      VARCHAR(20);

  SELECT alunos_embarcados, capacidade, percentual_ocupacao
    INTO   v_ocupados,       v_capacidade, v_percentual
  FROM   vw_ocupacao_rota
  WHERE  id_rota = p_id_rota;

  IF v_percentual >= 100 THEN
    SET v_nivel = 'LOTADO';
  ELSEIF v_percentual >= 80 THEN
    SET v_nivel = 'QUASE_LOTADO';
  ELSE
    SET v_nivel = 'NORMAL';
  END IF;

  INSERT INTO alerta_rota (id_rota, nivel_alerta, ocupacao_atual, capacidade, percentual, origem)
  VALUES (p_id_rota, v_nivel, v_ocupados, v_capacidade, v_percentual, 'TRIGGER');
END$$


-- -------------------------------------------------------------------------
-- 5) TRIGGER que integra Evento -> Procedure -> View -> Efeito persistente
--    trg_aluno_rota_ai_alerta
--    O que dispara: um INSERT em aluno_rota (um aluno acabou de ser
--    colocado em uma rota). Isso e diferente da trigger trg_aluno_rota_bi
--    do Trabalho 1 (que roda ANTES do insert so para bloquear lotacao);
--    esta roda DEPOIS do insert e serve para registrar o alerta.
-- -------------------------------------------------------------------------
CREATE TRIGGER trg_aluno_rota_ai_alerta
AFTER INSERT ON aluno_rota
FOR EACH ROW
BEGIN
  CALL sp_verificar_ocupacao_rota(NEW.id_rota);
END$$


-- -------------------------------------------------------------------------
-- 6) STORED PROCEDURE 1 (chamada pelo usuario/coordenador)
--    sp_registrar_status_ocupacao
--    Quando executa: sob demanda, quando o coordenador quer um raio-x de
--    todas as rotas de uma vez (ex.: CALL sp_registrar_status_ocupacao(80);)
--    O que faz: percorre (cursor) a View vw_ocupacao_rota linha a linha;
--    para cada rota cujo percentual de ocupacao for maior ou igual ao
--    limite informado, grava (efeito persistente) um alerta em
--    alerta_rota com origem = 'MANUAL'.
-- -------------------------------------------------------------------------
CREATE PROCEDURE sp_registrar_status_ocupacao (IN p_limite_alerta DECIMAL(5,2))
BEGIN
  DECLARE v_fim        INT DEFAULT 0;
  DECLARE v_id_rota    INT;
  DECLARE v_ocupados   INT;
  DECLARE v_capacidade INT;
  DECLARE v_percentual DECIMAL(5,2);
  DECLARE v_nivel      VARCHAR(20);

  DECLARE cur_rotas CURSOR FOR
    SELECT id_rota, alunos_embarcados, capacidade, percentual_ocupacao
    FROM   vw_ocupacao_rota;

  DECLARE CONTINUE HANDLER FOR NOT FOUND SET v_fim = 1;

  OPEN cur_rotas;

  loop_rotas: LOOP
    FETCH cur_rotas INTO v_id_rota, v_ocupados, v_capacidade, v_percentual;
    IF v_fim = 1 THEN
      LEAVE loop_rotas;
    END IF;

    IF v_percentual >= p_limite_alerta THEN
      IF v_percentual >= 100 THEN
        SET v_nivel = 'LOTADO';
      ELSE
        SET v_nivel = 'QUASE_LOTADO';
      END IF;

      INSERT INTO alerta_rota (id_rota, nivel_alerta, ocupacao_atual, capacidade, percentual, origem)
      VALUES (v_id_rota, v_nivel, v_ocupados, v_capacidade, v_percentual, 'MANUAL');
    END IF;
  END LOOP;

  CLOSE cur_rotas;
END$$

DELIMITER ;
