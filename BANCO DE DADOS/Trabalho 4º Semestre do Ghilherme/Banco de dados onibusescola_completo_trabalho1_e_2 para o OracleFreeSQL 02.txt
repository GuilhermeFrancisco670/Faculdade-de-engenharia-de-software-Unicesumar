-- Tabela de endereços (separada para atender a 3FN e não repetir dados)
CREATE TABLE endereco ( 
  id_endereco  NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, -- Chave primária que numera automaticamente
  cep          CHAR(8)       NOT NULL,                             -- CEP com tamanho fixo de 8 caracteres
  rua          VARCHAR2(100) NOT NULL,                             -- Nome da rua
  numero_casa  VARCHAR2(10)  NOT NULL,                             -- Número da residência
  bairro       VARCHAR2(60)  NULL,                                 -- Bairro
  cidade       VARCHAR2(60)  NOT NULL,                             -- Cidade
  uf           CHAR(2)       NOT NULL,                             -- Sigla do estado (PR, SP, etc.)
  descricao    VARCHAR2(150) NULL,                                 -- Ponto de referência ou detalhe
  CONSTRAINT chk_endereco_cep CHECK (LENGTH(cep) = 8)              -- Validação: CEP deve ter exatamente 8 dígitos
);

-- Tabela com dados das escolas
CREATE TABLE escola (
  id_escola        NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nome             VARCHAR2(100) NOT NULL,
  id_endereco      NUMBER        NOT NULL,                         -- Ligação com endereço (FK criada via ALTER TABLE)
  quantidade_aluno NUMBER        DEFAULT 0 NOT NULL,               -- Contador mantido por trigger
  turno            VARCHAR2(10)  NOT NULL,
  horario_entrada  VARCHAR2(8)   NOT NULL,                         -- Horário de entrada formato 'HH:MM:SS'
  horario_saida    VARCHAR2(8)   NOT NULL,                         -- Horário de saída formato 'HH:MM:SS'
  publico          CHAR(1)       DEFAULT 'S' NOT NULL,             -- 'S' para sim, 'N' para não
  CONSTRAINT chk_escola_turno   CHECK (turno IN ('MANHA','TARDE','NOITE','INTEGRAL')),
  CONSTRAINT chk_escola_publico CHECK (publico IN ('S','N')),
  CONSTRAINT chk_escola_qtd     CHECK (quantidade_aluno >= 0)
);

-- Tabela de responsáveis pelos alunos
CREATE TABLE responsavel (
  id_responsavel  NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nome            VARCHAR2(100) NOT NULL,
  numero_celular  VARCHAR2(15)  NOT NULL,
  id_endereco     NUMBER        NOT NULL,
  horario_em_casa VARCHAR2(8)   NULL,
  CONSTRAINT uq_responsavel_celular UNIQUE (numero_celular)        -- Não permite celulares duplicados
);

-- Tabela dos ônibus da frota escolar
CREATE TABLE onibus (
  id_onibus  NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  placa      CHAR(7)       NOT NULL,                               -- Placa única do veículo
  capacidade NUMBER        NOT NULL,                               -- Quantidade máxima de passageiros
  CONSTRAINT uq_onibus_placa UNIQUE (placa),
  CONSTRAINT chk_onibus_cap   CHECK (capacidade > 0)               -- Validação: capacidade deve ser positiva
);

-- Tabela de motoristas vinculados aos ônibus
CREATE TABLE motorista (
  id_motorista NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nome         VARCHAR2(100) NOT NULL,
  cnh          CHAR(11)      NOT NULL,
  telefone     VARCHAR2(15)  NULL,
  id_onibus    NUMBER        NULL,                                 -- Ônibus que o motorista dirige
  CONSTRAINT uq_motorista_cnh UNIQUE (cnh)
);

-- Tabela que define a rota do transporte escolar
CREATE TABLE rota (
  id_rota       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  descricao     VARCHAR2(150) NOT NULL,
  turno         VARCHAR2(10)  NOT NULL,
  horario_saida VARCHAR2(8)   NOT NULL,
  id_motorista  NUMBER        NOT NULL,
  id_escola     NUMBER        NOT NULL,
  CONSTRAINT chk_rota_turno CHECK (turno IN ('MANHA','TARDE','NOITE'))
);

-- Tabela principal de alunos
CREATE TABLE aluno (
  id_aluno        NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nome            VARCHAR2(100) NOT NULL,
  data_nascimento DATE          NULL,
  turno           VARCHAR2(10)  NOT NULL,
  id_escola       NUMBER        NOT NULL,
  id_responsavel  NUMBER        NOT NULL,
  id_endereco     NUMBER        NOT NULL,
  CONSTRAINT chk_aluno_turno CHECK (turno IN ('MANHA','TARDE','NOITE'))
);

-- Tabela associativa N:N (Muitos alunos para Muitas rotas) com Chave Primária Composta
CREATE TABLE aluno_rota (
  id_aluno       NUMBER NOT NULL,
  id_rota        NUMBER NOT NULL,
  ordem_embarque NUMBER NULL,                                      -- Posição do aluno na ordem da van/ônibus
  CONSTRAINT pk_aluno_rota PRIMARY KEY (id_aluno, id_rota)         -- Chave Primária composta (atende 2FN)
);

-- Tabela de auditoria para registrar matrículas e alterações
CREATE TABLE log_matricula (
  id_log     NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  id_aluno   NUMBER        NOT NULL,
  id_escola  NUMBER        NOT NULL,
  acao       VARCHAR2(20)  NOT NULL,
  data_hora  TIMESTAMP     DEFAULT CURRENT_TIMESTAMP NOT NULL     -- Grava o momento da ação
);

-- Tabela do Trabalho 2: Guarda os efeitos persistentes (alertas gerados por procedures/triggers)
CREATE TABLE alerta_rota (
  id_alerta       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  id_rota         NUMBER        NOT NULL,
  nivel_alerta    VARCHAR2(20)  NOT NULL,
  ocupacao_atual  NUMBER        NOT NULL,
  capacidade      NUMBER        NOT NULL,
  percentual      NUMBER(5,2)   NOT NULL,
  origem          VARCHAR2(10)  NOT NULL,                          -- 'MANUAL' ou 'TRIGGER'
  data_hora       TIMESTAMP     DEFAULT CURRENT_TIMESTAMP NOT NULL,
  CONSTRAINT chk_alerta_nivel  CHECK (nivel_alerta IN ('NORMAL','QUASE_LOTADO','LOTADO')),
  CONSTRAINT chk_alerta_origem CHECK (origem IN ('MANUAL','TRIGGER'))
);

-- VIEW 1: vw_ocupacao_rota
-- Reúne rota, motorista, ônibus e calcula ocupação e porcentagem
CREATE OR REPLACE VIEW vw_ocupacao_rota AS
SELECT
    r.id_rota,
    r.descricao                                             AS rota,
    r.turno,
    m.nome                                                  AS motorista,
    o.id_onibus,
    o.placa,
    o.capacidade,
    COUNT(ar.id_aluno)                                      AS alunos_embarcados,
    (o.capacidade - COUNT(ar.id_aluno))                     AS vagas_disponiveis,
    ROUND((COUNT(ar.id_aluno) / o.capacidade) * 100, 2)     AS percentual_ocupacao
FROM rota r
JOIN motorista m ON m.id_motorista = r.id_motorista
JOIN onibus o    ON o.id_onibus    = m.id_onibus
LEFT JOIN aluno_rota ar ON ar.id_rota = r.id_rota
GROUP BY r.id_rota, r.descricao, r.turno, m.nome, o.id_onibus, o.placa, o.capacidade;

-- VIEW 2: vw_ficha_embarque
-- Simplifica a consulta com 5 tabelas unidas para o motorista saber onde buscar cada aluno
CREATE OR REPLACE VIEW vw_ficha_embarque AS
SELECT
    ar.id_rota,
    ro.descricao                                              AS rota,
    ro.horario_saida,
    ar.ordem_embarque,
    a.id_aluno,
    a.nome                                                    AS aluno,
    en.rua || ', ' || en.numero_casa || NVL2(en.bairro, ' - ' || en.bairro, '') AS endereco_embarque,
    en.cep,
    resp.nome                                                 AS responsavel,
    resp.numero_celular,
    esc.nome                                                  AS escola
FROM aluno_rota ar
JOIN rota ro        ON ro.id_rota        = ar.id_rota
JOIN aluno a        ON a.id_aluno        = ar.id_aluno
JOIN endereco en    ON en.id_endereco    = a.id_endereco
JOIN responsavel resp ON resp.id_responsavel = a.id_responsavel
JOIN escola esc     ON esc.id_escola     = a.id_escola;

-- VIEW 3: vw_alunos_sem_rota
-- Lista os alunos matriculados que ainda não possuem rota atribuída
CREATE OR REPLACE VIEW vw_alunos_sem_rota AS
SELECT
    a.id_aluno,
    a.nome              AS aluno,
    a.turno,
    esc.nome            AS escola,
    resp.nome           AS responsavel,
    resp.numero_celular
FROM aluno a
JOIN escola esc       ON esc.id_escola      = a.id_escola
JOIN responsavel resp ON resp.id_responsavel = a.id_responsavel
LEFT JOIN aluno_rota ar ON ar.id_aluno = a.id_aluno
WHERE ar.id_aluno IS NULL;

-- TRIGGER 1 (Trabalho 1 - Auditoria e Atualização Automática)
-- Ao cadastrar um novo aluno, atualiza o contador na escola e gera um log
CREATE OR REPLACE TRIGGER trg_aluno_depois_inserir
AFTER INSERT ON aluno
FOR EACH ROW
BEGIN
  -- Atualiza o total de alunos na escola correspondente
  UPDATE escola
     SET quantidade_aluno = quantidade_aluno + 1
   WHERE id_escola = :NEW.id_escola;

  -- Grava o histórico na tabela de log
  INSERT INTO log_matricula (id_aluno, id_escola, acao)
  VALUES (:NEW.id_aluno, :NEW.id_escola, 'MATRICULA');
END;
/

-- STORED PROCEDURE 2 (Trabalho 2: Procedure chamada dentro de uma Trigger)
-- Lê a View vw_ocupacao_rota e grava um efeito persistente em alerta_rota
CREATE OR REPLACE PROCEDURE sp_verificar_ocupacao_rota (p_id_rota IN NUMBER)
IS
  v_ocupados   NUMBER;
  v_capacidade NUMBER;
  v_percentual NUMBER(5,2);
  v_nivel      VARCHAR2(20);
BEGIN
  -- Consulta dados prontos vindos diretamente da VIEW
  SELECT alunos_embarcados, capacidade, percentual_ocupacao
    INTO v_ocupados, v_capacidade, v_percentual
    FROM vw_ocupacao_rota
   WHERE id_rota = p_id_rota;

  -- Define a gravidade do alerta com base na ocupação
  IF v_percentual >= 100 THEN
    v_nivel := 'LOTADO';
  ELSIF v_percentual >= 80 THEN
    v_nivel := 'QUASE_LOTADO';
  ELSE
    v_nivel := 'NORMAL';
  END IF;

  -- Efeito persistente: Insere o alerta na tabela
  INSERT INTO alerta_rota (id_rota, nivel_alerta, ocupacao_atual, capacidade, percentual, origem)
  VALUES (p_id_rota, v_nivel, v_ocupados, v_capacidade, v_percentual, 'TRIGGER');
END;
/

-- TRIGGER 2 (Trabalho 2: Integração Evento -> Trigger -> Procedure -> View -> Tabela)
-- Dispara após um aluno ser adicionado em uma rota e aciona a procedure de verificação
CREATE OR REPLACE TRIGGER trg_aluno_rota_ai_alerta
AFTER INSERT ON aluno_rota
FOR EACH ROW
BEGIN
  -- Chama a Stored Procedure passando o ID da rota afetada
  sp_verificar_ocupacao_rota(:NEW.id_rota);
END;
/

-- STORED PROCEDURE 1 (Trabalho 2: Chamada pelo usuário/coordenador)
-- Lê a View vw_ocupacao_rota para todas as rotas e grava alertas manuais
CREATE OR REPLACE PROCEDURE sp_registrar_status_ocupacao (p_limite_alerta IN NUMBER)
IS
  v_nivel VARCHAR2(20);
BEGIN
  -- Percorre cada rota retornada pela VIEW
  FOR r IN (SELECT id_rota, alunos_embarcados, capacidade, percentual_ocupacao FROM vw_ocupacao_rota) LOOP
    -- Se a ocupação for maior ou igual ao parâmetro informado pelo usuário
    IF r.percentual_ocupacao >= p_limite_alerta THEN
      IF r.percentual_ocupacao >= 100 THEN
        v_nivel := 'LOTADO';
      ELSE
        v_nivel := 'QUASE_LOTADO';
      END IF;

      -- Grava o alerta na tabela (efeito persistente com origem MANUAL)
      INSERT INTO alerta_rota (id_rota, nivel_alerta, ocupacao_atual, capacidade, percentual, origem)
      VALUES (r.id_rota, v_nivel, r.alunos_embarcados, r.capacidade, r.percentual_ocupacao, 'MANUAL');
    END IF;
  END LOOP;
END;
/
