-- Tabela de endereços (separada para atender a 3FN e não duplicar dados de ruas)
CREATE TABLE endereco ( 
  id_endereco  INT          AUTO_INCREMENT PRIMARY KEY, -- Chave primária que numera sozinha
  cep          CHAR(8)      NOT NULL,                   -- CEP com tamanho fixo de 8 caracteres
  rua          VARCHAR(100) NOT NULL,                   -- Nome da rua
  numero_casa  VARCHAR(10)  NOT NULL,                   -- Número da casa
  bairro       VARCHAR(60)  NULL,                       -- Bairro
  cidade       VARCHAR(60)  NOT NULL,                   -- Nome da cidade
  uf           CHAR(2)      NOT NULL,                   -- Estado (PR, SP, etc.)
  descricao    VARCHAR(150) NULL                        -- Ponto de referência
);

-- Tabela com os dados das escolas
CREATE TABLE escola (
  id_escola        INT          AUTO_INCREMENT PRIMARY KEY,
  nome             VARCHAR(100) NOT NULL,
  id_endereco      INT          NOT NULL,
  quantidade_aluno INT          NOT NULL DEFAULT 0,     -- Total mantido automaticamente pela trigger
  turno            VARCHAR(10)  NOT NULL,
  horario_entrada  TIME         NOT NULL,               -- Horário de entrada
  horario_saida    TIME         NOT NULL,               -- Horário de saída
  publico          CHAR(1)      NOT NULL DEFAULT 'S',
  CONSTRAINT chk_escola_turno   CHECK (turno IN ('MANHA','TARDE','NOITE','INTEGRAL')),
  CONSTRAINT chk_escola_publico CHECK (publico IN ('S','N')),
  CONSTRAINT chk_escola_qtd     CHECK (quantidade_aluno >= 0)
);

-- Tabela de responsáveis pelos alunos
CREATE TABLE responsavel (
  id_responsavel  INT          AUTO_INCREMENT PRIMARY KEY,
  nome            VARCHAR(100) NOT NULL,
  numero_celular  VARCHAR(15)  NOT NULL,
  id_endereco     INT          NOT NULL,
  horario_em_casa TIME         NULL,
  CONSTRAINT uq_responsavel_celular UNIQUE (numero_celular) -- Celular único
);

-- Tabela de veículos da frota escolar
CREATE TABLE onibus (
  id_onibus  INT     AUTO_INCREMENT PRIMARY KEY,
  placa      CHAR(7) NOT NULL,
  capacidade INT     NOT NULL,
  CONSTRAINT uq_onibus_placa UNIQUE (placa),            -- Não permite placas repetidas
  CONSTRAINT chk_onibus_cap  CHECK (capacidade > 0)     -- Capacidade deve ser maior que zero
);

-- Tabela de motoristas vinculados aos veículos
CREATE TABLE motorista (
  id_motorista INT          AUTO_INCREMENT PRIMARY KEY,
  nome         VARCHAR(100) NOT NULL,
  cnh          CHAR(11)     NOT NULL,
  telefone     VARCHAR(15)  NULL,
  id_onibus    INT          NULL,
  CONSTRAINT uq_motorista_cnh UNIQUE (cnh)
);

-- Tabela de rotas do transporte escolar
CREATE TABLE rota (
  id_rota       INT          AUTO_INCREMENT PRIMARY KEY,
  descricao     VARCHAR(150) NOT NULL,
  turno         VARCHAR(10)  NOT NULL,
  horario_saida TIME         NOT NULL,
  id_motorista  INT          NOT NULL,
  id_escola     INT          NOT NULL,
  CONSTRAINT chk_rota_turno CHECK (turno IN ('MANHA','TARDE','NOITE'))
);

-- Tabela principal de alunos
CREATE TABLE aluno (
  id_aluno        INT          AUTO_INCREMENT PRIMARY KEY,
  nome            VARCHAR(100) NOT NULL,
  data_nascimento DATE         NULL,
  turno           VARCHAR(10)  NOT NULL,
  id_escola       INT          NOT NULL,
  id_responsavel  INT          NOT NULL,
  id_endereco     INT          NOT NULL,
  CONSTRAINT chk_aluno_turno CHECK (turno IN ('MANHA','TARDE','NOITE'))
);

-- Tabela associativa N:N (Muitos alunos para Muitas rotas - atende a 2FN)
CREATE TABLE aluno_rota (
  id_aluno       INT NOT NULL,
  id_rota        INT NOT NULL,
  ordem_embarque INT NULL,
  PRIMARY KEY (id_aluno, id_rota) -- Chave primária composta
);

-- Tabela de auditoria para registrar as matrículas
CREATE TABLE log_matricula (
  id_log     INT         AUTO_INCREMENT PRIMARY KEY,
  id_aluno   INT         NOT NULL,
  id_escola  INT         NOT NULL,
  acao       VARCHAR(20) NOT NULL,
  data_hora  DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Tabela do Trabalho 2: Guarda os efeitos persistentes (alertas gerados)
CREATE TABLE alerta_rota (
  id_alerta       INT AUTO_INCREMENT PRIMARY KEY,
  id_rota         INT NOT NULL,
  nivel_alerta    VARCHAR(20)  NOT NULL,
  ocupacao_atual  INT          NOT NULL,
  capacidade      INT          NOT NULL,
  percentual      DECIMAL(5,2) NOT NULL,
  origem          VARCHAR(10)  NOT NULL, -- 'MANUAL' ou 'TRIGGER'
  data_hora       DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT chk_alerta_nivel  CHECK (nivel_alerta IN ('NORMAL','QUASE_LOTADO','LOTADO')),
  CONSTRAINT chk_alerta_origem CHECK (origem IN ('MANUAL','TRIGGER'))
);

-- VIEW 1: vw_ocupacao_rota
-- Reúne rota, motorista, ônibus e calcula vagas e percentual de ocupação
CREATE VIEW vw_ocupacao_rota AS
SELECT
    r.id_rota,
    r.descricao                                           AS rota,
    r.turno,
    m.nome                                                AS motorista,
    o.id_onibus,
    o.placa,
    o.capacidade,
    COUNT(ar.id_aluno)                                    AS alunos_embarcados,
    (o.capacidade - COUNT(ar.id_aluno))                   AS vagas_disponiveis,
    ROUND((COUNT(ar.id_aluno) / o.capacidade) * 100, 2)   AS percentual_ocupacao
FROM rota r
JOIN motorista m ON m.id_motorista = r.id_motorista
JOIN onibus o    ON o.id_onibus    = m.id_onibus
LEFT JOIN aluno_rota ar ON ar.id_rota = r.id_rota
GROUP BY r.id_rota, r.descricao, r.turno, m.nome, o.id_onibus, o.placa, o.capacidade;

-- VIEW 2: vw_ficha_embarque
-- Reúne as 5 tabelas para facilitar a consulta diária do motorista
CREATE VIEW vw_ficha_embarque AS
SELECT
    ar.id_rota,
    ro.descricao                                          AS rota,
    ro.horario_saida,
    ar.ordem_embarque,
    a.id_aluno,
    a.nome                                                AS aluno,
    CONCAT(en.rua, ', ', en.numero_casa, IFNULL(CONCAT(' - ', en.bairro), '')) AS endereco_embarque,
    en.cep,
    resp.nome                                             AS responsavel,
    resp.numero_celular,
    esc.nome                                              AS escola
FROM aluno_rota ar
JOIN rota ro          ON ro.id_rota        = ar.id_rota
JOIN aluno a          ON a.id_aluno        = ar.id_aluno
JOIN endereco en      ON en.id_endereco    = a.id_endereco
JOIN responsavel resp ON resp.id_responsavel = a.id_responsavel
JOIN escola esc       ON esc.id_escola     = a.id_escola;

-- VIEW 3: vw_alunos_sem_rota
-- Mostra os alunos cadastrados que ainda não foram colocados em nenhum ônibus
CREATE VIEW vw_alunos_sem_rota AS
SELECT
    a.id_aluno,
    a.nome              AS aluno,
    a.turno,
    esc.nome            AS escola,
    resp.nome           AS responsavel,
    resp.numero_celular
FROM aluno a
JOIN escola esc       ON esc.id_escola      = a.id_escola
JOIN responsavel resp ON resp.id_responsavel = a.id_responsavel
LEFT JOIN aluno_rota ar ON ar.id_aluno = a.id_aluno
WHERE ar.id_aluno IS NULL;

DELIMITER $$

-- TRIGGER 1 (Trabalho 1 - Auditoria e Atualização Automática)
-- Ao inserir um aluno, atualiza o contador na escola e grava um registro de log
CREATE TRIGGER trg_aluno_depois_inserir
AFTER INSERT ON aluno
FOR EACH ROW
BEGIN
  -- Atualiza o total de alunos na tabela escola
  UPDATE escola
     SET quantidade_aluno = quantidade_aluno + 1
   WHERE id_escola = NEW.id_escola;

  -- Grava o histórico de matrícula
  INSERT INTO log_matricula (id_aluno, id_escola, acao)
  VALUES (NEW.id_aluno, NEW.id_escola, 'MATRICULA');
END$$


-- STORED PROCEDURE 2 (Trabalho 2: Procedure chamada pela Trigger)
-- Lê a View vw_ocupacao_rota e grava um efeito persistente em alerta_rota
CREATE PROCEDURE sp_verificar_ocupacao_rota (IN p_id_rota INT)
BEGIN
  DECLARE v_ocupados   INT;
  DECLARE v_capacidade INT;
  DECLARE v_percentual DECIMAL(5,2);
  DECLARE v_nivel      VARCHAR(20);

  -- Busca as informações diretamente da VIEW
  SELECT alunos_embarcados, capacidade, percentual_ocupacao
    INTO v_ocupados, v_capacidade, v_percentual
    FROM vw_ocupacao_rota
   WHERE id_rota = p_id_rota;

  -- Define o nível de alerta
  IF v_percentual >= 100 THEN
    SET v_nivel = 'LOTADO';
  ELSEIF v_percentual >= 80 THEN
    SET v_nivel = 'QUASE_LOTADO';
  ELSE
    SET v_nivel = 'NORMAL';
  END IF;

  -- Efeito persistente: Insere o registro de alerta
  INSERT INTO alerta_rota (id_rota, nivel_alerta, ocupacao_atual, capacidade, percentual, origem)
  VALUES (p_id_rota, v_nivel, v_ocupados, v_capacidade, v_percentual, 'TRIGGER');
END$$


-- TRIGGER 2 (Trabalho 2: Integração Evento -> Trigger -> Procedure -> View -> Tabela)
-- Dispara após um aluno ser colocado em uma rota e aciona a Procedure 2
CREATE TRIGGER trg_aluno_rota_ai_alerta
AFTER INSERT ON aluno_rota
FOR EACH ROW
BEGIN
  -- Aciona a procedure para recalcular e gerar o alerta
  CALL sp_verificar_ocupacao_rota(NEW.id_rota);
END$$


-- STORED PROCEDURE 1 (Trabalho 2: Chamada sob demanda pelo usuário)
-- Percorre a View vw_ocupacao_rota e grava alertas manuais para rotas acima do limite
CREATE PROCEDURE sp_registrar_status_ocupacao (IN p_limite_alerta DECIMAL(5,2))
BEGIN
  DECLARE v_fim        INT DEFAULT 0;
  DECLARE v_id_rota    INT;
  DECLARE v_ocupados   INT;
  DECLARE v_capacidade INT;
  DECLARE v_percentual DECIMAL(5,2);
  DECLARE v_nivel      VARCHAR(20);

  -- Declaração do Cursor sobre a View
  DECLARE cur_rotas CURSOR FOR
    SELECT id_rota, alunos_embarcados, capacidade, percentual_ocupacao
    FROM vw_ocupacao_rota;

  DECLARE CONTINUE HANDLER FOR NOT FOUND SET v_fim = 1;

  OPEN cur_rotas;

  loop_rotas: LOOP
    FETCH cur_rotas INTO v_id_rota, v_ocupados, v_capacidade, v_percentual;
    IF v_fim = 1 THEN
      LEAVE loop_rotas;
    END IF;

    -- Se o percentual atingir o limite estipulado pelo usuário
    IF v_percentual >= p_limite_alerta THEN
      IF v_percentual >= 100 THEN
        SET v_nivel = 'LOTADO';
      ELSE
        SET v_nivel = 'QUASE_LOTADO';
      END IF;

      -- Grava o alerta na tabela
      INSERT INTO alerta_rota (id_rota, nivel_alerta, ocupacao_atual, capacidade, percentual, origem)
      VALUES (v_id_rota, v_nivel, v_ocupados, v_capacidade, v_percentual, 'MANUAL');
    END IF;
  END LOOP;

  CLOSE cur_rotas;
END$$

DELIMITER ;

INNER JOIN motorista m ON m.id_motorista = r.id_motorista
INNER JOIN onibus o ON o.id_onibus = m.id_onibus;
